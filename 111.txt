<?php include "includes/includes.php";

// Fetch filters
$filterCustomer = $_REQUEST['customer'] ?? '';
$filterJournal  = $_REQUEST['journal'] ?? '';
$filterStage    = $_REQUEST['stage'] ?? '';
$filterFromDate = $_REQUEST['from_date'] ?? '';
$filterToDate   = $_REQUEST['to_date'] ?? '';
$filterMonth    = $_REQUEST['mfrom_dt'] ?? '';
$dateWise       = $_REQUEST['date_wise'] ?? '';

// Build WHERE conditions
$where = [];

// Customer
if ($filterCustomer && $filterCustomer != 'all') {
    $where[] = "cust_id = ".intval($filterCustomer);
}

if ($filterJournal && $filterJournal != 'all') {
    $where[] = "j_id = ".intval($filterJournal);
}
// Stage
if ($filterStage) {
    if ($filterStage == 'FP') {
        $where[] = "stage = 'FP'";
    } else {
        $where[] = "stage LIKE '".$db->real_escape_string($filterStage)."%' ";
    }
}

// Convert day-wise dates
if ($filterFromDate && $filterToDate) {

    $fromParts = explode('-', $filterFromDate);
    $toParts   = explode('-', $filterToDate);

    $from = "{$fromParts[2]}-{$fromParts[1]}-{$fromParts[0]}";
    $to   = "{$toParts[2]}-{$toParts[1]}-{$toParts[0]}";

    $where[] = "DATE(
        COALESCE(
            CASE WHEN stage='FP' THEN recv_dt ELSE received_date END,
            recv_dt
        )
    ) BETWEEN '{$from}' AND '{$to}'";
}

// Month filter
if ($dateWise === "m" && !empty($filterMonth)) {

    $monthParts = explode('-', $filterMonth);
    $month = intval($monthParts[0]);
    $year  = intval($monthParts[1]);

    $where[] = "
        MONTH(
            COALESCE(
                CASE WHEN stage='FP' THEN recv_dt ELSE received_date END,
                recv_dt
            )
        ) = {$month}

        AND 

        YEAR(
            COALESCE(
                CASE WHEN stage='FP' THEN recv_dt ELSE received_date END,
                recv_dt
            )
        ) = {$year}
    ";
}

$whereSql = $where ? "WHERE " . implode(" AND ", $where) : "";


$baseQuery = "

(
    /* ARCHIVE DATA (Priority) */
    SELECT 
        d.a_id,
        d.stage,
        d.sent_date,
        i.recv_dt,
        r.received_date,
        c.id AS cust_id,
        c.cust_name,
        j.j_id,
        j.j_code
    FROM workflow_archive.inw_dispatch_history d
    JOIN workflow_archive.inw_inward_dtl i 
        ON d.a_id = i.id
    LEFT JOIN workflow_archive.inw_revisions_dtl r 
        ON d.a_id = r.inw_id
    JOIN workflow_archive.adm_customer_master c 
        ON i.cust_id = c.id
    JOIN workflow_archive.adm_journals j 
        ON i.j_id = j.j_id

    UNION ALL

    /* LIVE DATA (ONLY IF NOT IN ARCHIVE) */
    SELECT 
        d.a_id,
        d.stage,
        d.sent_date,
        i.recv_dt,
        r.received_date,
        c.id AS cust_id,
        c.cust_name,
        j.j_id,
        j.j_code
    FROM workflow.inw_dispatch_history d
    JOIN workflow.inw_inward_dtl i 
        ON d.a_id = i.id
    LEFT JOIN workflow.inw_revisions_dtl r 
        ON d.a_id = r.inw_id
    JOIN workflow.adm_customer_master c 
        ON i.cust_id = c.id
    JOIN workflow.adm_journals j 
        ON i.j_id = j.j_id

    WHERE NOT EXISTS (
        SELECT 1
        FROM workflow_archive.inw_dispatch_history ad
        WHERE ad.a_id = d.a_id
    )

) AS dispatch_data
";
$sql = "

SELECT 
    cust_name,
    j_code,

    CASE 
        WHEN stage = 'FP' THEN 'First Proof'
        WHEN stage LIKE 'REV%' THEN stage
        WHEN stage LIKE 'FIN%' THEN stage
        WHEN stage LIKE 'PMC%' THEN stage
        WHEN stage LIKE 'DRAFT%' THEN stage
        ELSE stage
    END AS stage_name,

    CASE
        WHEN DATEDIFF(
            sent_date,
            COALESCE(
                CASE WHEN stage='FP' THEN recv_dt ELSE received_date END,
                recv_dt
            )
        ) BETWEEN 0 AND 1 THEN 'Day 1'

        WHEN DATEDIFF(
            sent_date,
            COALESCE(
                CASE WHEN stage='FP' THEN recv_dt ELSE received_date END,
                recv_dt
            )
        ) = 2 THEN 'Day 2'

        WHEN DATEDIFF(
            sent_date,
            COALESCE(
                CASE WHEN stage='FP' THEN recv_dt ELSE received_date END,
                recv_dt
            )
        ) = 3 THEN 'Day 3'

        WHEN DATEDIFF(
            sent_date,
            COALESCE(
                CASE WHEN stage='FP' THEN recv_dt ELSE received_date END,
                recv_dt
            )
        ) = 4 THEN 'Day 4'

        WHEN DATEDIFF(
            sent_date,
            COALESCE(
                CASE WHEN stage='FP' THEN recv_dt ELSE received_date END,
                recv_dt
            )
        ) = 5 THEN 'Day 5'

        WHEN DATEDIFF(
            sent_date,
            COALESCE(
                CASE WHEN stage='FP' THEN recv_dt ELSE received_date END,
                recv_dt
            )
        ) = 6 THEN 'Day 6'

        WHEN DATEDIFF(
            sent_date,
            COALESCE(
                CASE WHEN stage='FP' THEN recv_dt ELSE received_date END,
                recv_dt
            )
        ) = 7 THEN 'Day 7'

        WHEN DATEDIFF(
            sent_date,
            COALESCE(
                CASE WHEN stage='FP' THEN recv_dt ELSE received_date END,
                recv_dt
            )
        ) = 8 THEN 'Day 8'

        WHEN DATEDIFF(
            sent_date,
            COALESCE(
                CASE WHEN stage='FP' THEN recv_dt ELSE received_date END,
                recv_dt
            )
        ) = 9 THEN 'Day 9'

        WHEN DATEDIFF(
            sent_date,
            COALESCE(
                CASE WHEN stage='FP' THEN recv_dt ELSE received_date END,
                recv_dt
            )
        ) = 10 THEN 'Day 10'

        ELSE '>10 Days'
    END AS dispatch_days_bucket,

    COUNT(*) AS dispatch_count

FROM $baseQuery

$whereSql

GROUP BY 
    cust_name,
    j_code,
    stage_name,
    dispatch_days_bucket

ORDER BY 
    cust_name,
    j_code,
    stage_name

";

// Render function
function renderStageTable($stageGroup, $data, $dayBuckets) {
    $tableId = strtolower($stageGroup) . "_table";
    $hasData = false;

    // Check if this stageGroup has any data
    foreach ($data as $stage => $customers) {
        if ($stageGroup == 'FP' && $stage == 'First Proof') $hasData = true;
        if ($stageGroup == 'REV' && strpos($stage, 'REV') === 0) $hasData = true;
        if ($stageGroup == 'FIN' && strpos($stage, 'FIN') === 0) $hasData = true;
        if ($stageGroup == 'PMC' && strpos($stage, 'PMC') === 0) $hasData = true;
        if ($stageGroup == 'DRAFT' && strpos($stage, 'DRAFT') === 0) $hasData = true;
    }

    if (!$hasData) return;

    echo "<h3 class='mt-5 mb-3 text-primary'>{$stageGroup}</h3>";
    echo "<table id='{$tableId}' class='table table-bordered table-sticky display nowrap' style='width:100%'>";
    echo "<thead class='table-dark'><tr><th>Customer</th><th>Journal</th><th>Stage</th>";
    foreach ($dayBuckets as $bucket) echo "<th>{$bucket}</th>";
    echo "<th>Total</th></tr></thead><tbody>";

    // Data rows
    foreach ($data as $stage => $customers) {
        if ($stageGroup == 'FP' && $stage != 'First Proof') continue;
        if ($stageGroup == 'REV' && strpos($stage, 'REV') !== 0) continue;
        if ($stageGroup == 'FIN' && strpos($stage, 'FIN') !== 0) continue;
        if ($stageGroup == 'PMC' && strpos($stage, 'PMC') !== 0) continue;
        if ($stageGroup == 'DRAFT' && strpos($stage, 'DRAFT') !== 0) continue;

        foreach ($customers as $cust => $journals) {
            foreach ($journals as $journal => $buckets) {
                $rowTotal = 0;
                echo "<tr><td>{$cust}</td><td>{$journal}</td><td>{$stage}</td>";
                foreach ($dayBuckets as $bucket) {
                    $val = $buckets[$bucket] ?? 0;
                    echo "<td>{$val}</td>";
                    $rowTotal += $val;
                }
                echo "<td>{$rowTotal}</td>";
                echo "</tr>";
            }
        }
    }

    // Totals row (filtered by stageGroup)
    $totals = array_fill_keys($dayBuckets, 0);
    $grandTotal = 0;

    foreach ($data as $stage => $customers) {
        if ($stageGroup == 'FP' && $stage != 'First Proof') continue;
        if ($stageGroup == 'REV' && strpos($stage, 'REV') !== 0) continue;
        if ($stageGroup == 'FIN' && strpos($stage, 'FIN') !== 0) continue;
        if ($stageGroup == 'PMC' && strpos($stage, 'PMC') !== 0) continue;
        if ($stageGroup == 'DRAFT' && strpos($stage, 'DRAFT') !== 0) continue;

        foreach ($customers as $cust => $journals) {
            foreach ($journals as $journal => $buckets) {
                foreach ($dayBuckets as $bucket) {
                    $totals[$bucket] += $buckets[$bucket] ?? 0;
                    $grandTotal += $buckets[$bucket] ?? 0;
                }
            }
        }
    }

    echo "<tr class='table-secondary fw-bold'><td colspan='3' class='text-end'>Total</td>";
    foreach ($dayBuckets as $bucket) echo "<td>{$totals[$bucket]}</td>";
    echo "<td>{$grandTotal}</td>";
    echo "</tr>";

    echo "</tbody></table><hr>";
}
